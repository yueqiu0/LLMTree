# IOT (LLM) 命令

## 1. airline-satisfaction (2分类，测试集100，深度4)
conda activate qy_env && python3 evaluate.py \
  --config config/evaluate/llm_gen_directly-airline-satisfaction.yml \
  --tree-type LLM \
  --with-llm 0 \
  --train-sizes 4 \
  --max-depth 4 \
  --num-tests-per-set 10 \

  --exp-name baseline_llm_directly_airline-satisfaction
